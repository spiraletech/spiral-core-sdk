cmake_minimum_required(VERSION 3.20)
project(spiral_core_installed_consumer LANGUAGES CXX)
find_package(SpiralCore 0.2 CONFIG REQUIRED)
if(NOT TARGET Spiral::Core)
 message(FATAL_ERROR "Installed package must export Spiral::Core")
endif()
add_executable(spiral_installed_smoke main.cpp)
target_link_libraries(spiral_installed_smoke PRIVATE Spiral::Core)
include(CTest)
if(BUILD_TESTING)
 add_test(NAME spiral.installed.consumer COMMAND spiral_installed_smoke)
endif()
